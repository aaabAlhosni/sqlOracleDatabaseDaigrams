  TASK 1 │ Simple One-Time Scheduler Job

-- Create JOB_GREET_EMPLOYEES (fires 2 minutes from now)
BEGIN
    DBMS_SCHEDULER.CREATE_JOB(
        job_name   => 'JOB_GREET_EMPLOYEES',
        job_type   => 'PLSQL_BLOCK',
        job_action => '
            BEGIN
                DBMS_OUTPUT.PUT_LINE(''Payroll System Initialized'');
                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL, ''Payroll System Initialized'', SYSTIMESTAMP);
                COMMIT;
            END;',
        start_date => SYSTIMESTAMP + INTERVAL ''2'' MINUTE,
        enabled    => TRUE,
        comments   => 'One-time greeting job for system initialization'
    );
END;
/


-- Verify job state
SELECT job_name, job_type, state, last_start_date
FROM USER_SCHEDULER_JOBS
WHERE job_name = 'JOB_GREET_EMPLOYEES';

-- After 2 minutes: confirm execution status
SELECT log_id, job_name, status, log_date, additional_info
FROM USER_SCHEDULER_JOB_LOG
WHERE job_name = 'JOB_GREET_EMPLOYEES'
ORDER BY log_date DESC;

-- Confirm the log entry in EMPLOYEE_LOG
SELECT * FROM EMPLOYEE_LOG
WHERE action = 'Payroll System Initialized'
ORDER BY log_timestamp DESC;

  TASK 2 │ Recurring Job — Daily Leave Report
────────────────────────────────────────────────────────────

-- Create JOB_DAILY_LEAVE_REPORT (daily at 07:00 AM)
BEGIN
    DBMS_SCHEDULER.CREATE_JOB(
        job_name        => 'JOB_DAILY_LEAVE_REPORT',
        job_type        => 'PLSQL_BLOCK',
        job_action      => '
            DECLARE
                v_count NUMBER;
            BEGIN
                SELECT COUNT(*) INTO v_count
                FROM LEAVEE
                WHERE TRUNC(leave_date) = TRUNC(SYSDATE);

                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL,
                    ''DAILY LEAVE REPORT: '' || v_count || '' records today'',
                    SYSTIMESTAMP);
                COMMIT;
            END;',
        start_date      => TRUNC(SYSDATE + 1) + 7/24,
        repeat_interval => 'FREQ=DAILY; BYHOUR=7; BYMINUTE=0; BYSECOND=0',
        enabled         => TRUE,
        comments        => 'Daily leave count summary inserted into EMPLOYEE_LOG'
    );
END;
/


-- Show job definition
SELECT job_name, state, repeat_interval, next_run_date
FROM USER_SCHEDULER_JOBS
WHERE job_name = 'JOB_DAILY_LEAVE_REPORT';

  TASK 4 │ Job Error Handling & Notifications
────────────────────────────────────────────────────────────

-- (a) Wrap job action in BEGIN...EXCEPTION...END
BEGIN
    DBMS_SCHEDULER.DISABLE('JOB_DAILY_LEAVE_REPORT');

    DBMS_SCHEDULER.SET_ATTRIBUTE(
        name      => 'JOB_DAILY_LEAVE_REPORT',
        attribute => 'job_action',
        value     => '
            DECLARE
                v_count NUMBER;
            BEGIN
                SELECT COUNT(*) INTO v_count
                FROM LEAVEE
                WHERE TRUNC(leave_date) = TRUNC(SYSDATE);

                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL,
                    ''DAILY LEAVE REPORT: ''||v_count||'' records today'',
                    SYSTIMESTAMP);
                COMMIT;
            EXCEPTION
                WHEN OTHERS THEN
                    INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                    VALUES (NULL, ''JOB_ERROR: '' || SQLERRM, SYSTIMESTAMP);
                    COMMIT;
                    RAISE;
            END;'
    );

    DBMS_SCHEDULER.ENABLE('JOB_DAILY_LEAVE_REPORT');
END;
/


-- (b) Set max_failures = 3 (auto-disables after 3 consecutive failures)
BEGIN
    DBMS_SCHEDULER.SET_ATTRIBUTE(
        name      => 'JOB_DAILY_LEAVE_REPORT',
        attribute => 'max_failures',
        value     => 3
    );
END;
/


-- (c) Force a failure by referencing a non-existent table
BEGIN
    DBMS_SCHEDULER.DISABLE('JOB_DAILY_LEAVE_REPORT');

    DBMS_SCHEDULER.SET_ATTRIBUTE(
        name      => 'JOB_DAILY_LEAVE_REPORT',
        attribute => 'job_action',
        value     => '
            DECLARE v_count NUMBER;
            BEGIN
                SELECT COUNT(*) INTO v_count FROM NON_EXISTENT_TABLE_XYZ;
            EXCEPTION
                WHEN OTHERS THEN
                    INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                    VALUES (NULL, ''JOB_ERROR: '' || SQLERRM, SYSTIMESTAMP);
                    COMMIT;
                    RAISE;
            END;'
    );

    DBMS_SCHEDULER.ENABLE('JOB_DAILY_LEAVE_REPORT');
    DBMS_SCHEDULER.RUN_JOB('JOB_DAILY_LEAVE_REPORT');
END;
/

-- Check failure in scheduler log
SELECT log_id, job_name, status, error#, log_date

WHERE job_name = 'JOB_DAILY_LEAVE_REPORT'
ORDER BY log_date DESC;

-- Check JOB_ERROR entry in EMPLOYEE_LOG
SELECT * FROM EMPLOYEE_LOG

  TASK 5 │ Job Chain — Multi-Step Payroll Pipeline
────────────────────────────────────────────────────────────
-- Prerequisites: EMPLOYEE_LOG, SALARY_AUDIT, MVW_MONTHLY_PAYROLL,
-- and SP_PROCESS_PAYROLL must all exist before compiling the chain.

-- Create step 1 program: process payroll for all departments
BEGIN
    DBMS_SCHEDULER.CREATE_PROGRAM(
        program_name   => 'PROG_STEP_PROCESS',
        program_type   => 'PLSQL_BLOCK',
        program_action => '
            DECLARE
                v_count NUMBER := 0;
            BEGIN
                FOR dept IN (SELECT DISTINCT job_id FROM EMPLOYEE) LOOP
                    SP_PROCESS_PAYROLL(dept.job_id, TRUNC(LAST_DAY(SYSDATE)));
                    v_count := v_count + 1;
                END LOOP;
                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL,
                    ''STEP_PROCESS: '' || v_count || '' departments processed'',
                    SYSTIMESTAMP);
                COMMIT;
            END;',
        enabled => TRUE
    );
END;
/


-- Create step 2 program: audit payroll totals
BEGIN
    DBMS_SCHEDULER.CREATE_PROGRAM(
        program_name   => 'PROG_STEP_AUDIT',
        program_type   => 'PLSQL_BLOCK',
        program_action => '
            DECLARE
                v_total NUMBER;
            BEGIN
                SELECT SUM(total_am) INTO v_total
                FROM PAYROLL
                WHERE TRUNC(pay_date) = TRUNC(LAST_DAY(SYSDATE));

                INSERT INTO SALARY_AUDIT (emp_id, old_amount, new_amount)
                VALUES (NULL, 0, NVL(v_total, 0));

                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL,
                    ''STEP_AUDIT: total payroll = '' || NVL(v_total, 0),
                    SYSTIMESTAMP);
                COMMIT;
            END;',
        enabled => TRUE
    );
END;
/


-- Create step 3 program: refresh materialized view
BEGIN
    DBMS_SCHEDULER.CREATE_PROGRAM(
        program_name   => 'PROG_STEP_REPORT',
        program_type   => 'PLSQL_BLOCK',
        program_action => '
            BEGIN
                DBMS_MVIEW.REFRESH(''MVW_MONTHLY_PAYROLL'');
                INSERT INTO EMPLOYEE_LOG (emp_id, action, log_timestamp)
                VALUES (NULL, ''STEP_REPORT: MVW_MONTHLY_PAYROLL refreshed'',
                        SYSTIMESTAMP);
                COMMIT;
            END;',
        enabled => TRUE
    );
END;
/


-- Create the chain object
BEGIN
    DBMS_SCHEDULER.CREATE_CHAIN(
        chain_name => 'CHAIN_PAYROLL_PIPELINE',
        comments   => 'End-of-month 3-step payroll pipeline'
    );
END;
/


-- Add steps to the chain
BEGIN
    DBMS_SCHEDULER.DEFINE_CHAIN_STEP(
        chain_name   => 'CHAIN_PAYROLL_PIPELINE',
        step_name    => 'STEP_PROCESS',
        program_name => 'PROG_STEP_PROCESS'
    );
    DBMS_SCHEDULER.DEFINE_CHAIN_STEP(
        chain_name   => 'CHAIN_PAYROLL_PIPELINE',
        step_name    => 'STEP_AUDIT',
        program_name => 'PROG_STEP_AUDIT'
    );
    DBMS_SCHEDULER.DEFINE_CHAIN_STEP(
        chain_name   => 'CHAIN_PAYROLL_PIPELINE',
        step_name    => 'STEP_REPORT',
        program_name => 'PROG_STEP_REPORT'
    );
END;
/


-- Define conditional chain rules
BEGIN
    DBMS_SCHEDULER.DEFINE_CHAIN_RULE(
        chain_name => 'CHAIN_PAYROLL_PIPELINE',
        rule_name  => 'RULE_START',
        condition  => 'TRUE',
        action     => 'START STEP_PROCESS'
    );
    DBMS_SCHEDULER.DEFINE_CHAIN_RULE(
        chain_name => 'CHAIN_PAYROLL_PIPELINE',
        rule_name  => 'RULE_AUDIT',
        condition  => 'STEP_PROCESS COMPLETED SUCCESSFULLY',
        action     => 'START STEP_AUDIT'
    );
    DBMS_SCHEDULER.DEFINE_CHAIN_RULE(
        chain_name => 'CHAIN_PAYROLL_PIPELINE',
        rule_name  => 'RULE_REPORT',
        condition  => 'STEP_AUDIT COMPLETED SUCCESSFULLY',
        action     => 'START STEP_REPORT'
    );
    DBMS_SCHEDULER.DEFINE_CHAIN_RULE(
        chain_name => 'CHAIN_PAYROLL_PIPELINE',
        rule_name  => 'RULE_END',
        condition  => 'STEP_REPORT COMPLETED SUCCESSFULLY',
        action     => 'END'
    );
END;
/


-- Enable the chain
BEGIN
    DBMS_SCHEDULER.ENABLE('CHAIN_PAYROLL_PIPELINE');
END;
/


-- Create job to run chain on last day of each month at 23:00
BEGIN
    DBMS_SCHEDULER.CREATE_JOB(
        job_name        => 'JOB_PAYROLL_CHAIN',
        job_type        => 'CHAIN',
        job_action      => 'CHAIN_PAYROLL_PIPELINE',
