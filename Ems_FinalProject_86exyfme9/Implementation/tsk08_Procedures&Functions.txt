-- Create SP_ADD_EMPLOYEE
CREATE OR REPLACE PROCEDURE SP_ADD_EMPLOYEE (
    p_emp_pass  IN VARCHAR2,
    p_age       IN NUMBER,
    p_gender    IN CHAR,
    p_F_Name    IN VARCHAR2,
    p_L_Name    IN VARCHAR2,
    p_street    IN VARCHAR2,
    p_zip_code  IN VARCHAR2,
    p_job_id    IN NUMBER,
    p_salary_id IN NUMBER,
    p_email     IN VARCHAR2
) AS
    v_count            NUMBER;
    v_emp_id           NUMBER;
    ex_duplicate_email EXCEPTION;
BEGIN
    -- Validate email uniqueness
    SELECT COUNT(*) INTO v_count
    FROM EMP_EMAIL WHERE LOWER(emp_email) = LOWER(p_email);

    IF v_count > 0 THEN
        RAISE ex_duplicate_email;
    END IF;

    -- Sequence the PK and insert employee
    v_emp_id := seq_employee.NEXTVAL;

    INSERT INTO EMPLOYEE (emp_id, emp_pass, age, gender, F_Name, L_Name,
                          street, zip_code, job_id, salary_id)
    VALUES (v_emp_id, p_emp_pass, p_age, p_gender, p_F_Name, p_L_Name,
            p_street, p_zip_code, p_job_id, p_salary_id);

    INSERT INTO EMP_EMAIL (emp_email, emp_id)
    VALUES (LOWER(p_email), v_emp_id);

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Employee ' || p_F_Name || ' ' || p_L_Name
        || ' added successfully with ID ' || v_emp_id || '.');

EXCEPTION
    WHEN ex_duplicate_email THEN
        DBMS_OUTPUT.PUT_LINE('Error: email ' || p_email
            || ' is already in use. Employee not added.');
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Unexpected error: ' || SQLERRM);
END SP_ADD_EMPLOYEE;

  TASK 2 │ Function — Calculate Net Salary
────────────────────────────────────────────────────────────

-- Create FN_NET_SALARY
CREATE OR REPLACE FUNCTION FN_NET_SALARY (
    p_emp_id IN NUMBER
) RETURN NUMBER AS
    v_net NUMBER;
BEGIN
    SELECT sb.amount + NVL(sb.bons, 0)
    INTO v_net
    FROM EMPLOYEE e
    JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
    WHERE e.emp_id = p_emp_id;

    RETURN v_net;

EXCEPTION
    WHEN NO_DATA_FOUND THEN RETURN NULL;
    WHEN OTHERS       THEN RETURN NULL;
END FN_NET_SALARY;
/


-- Use in SELECT — all employees ordered by net salary highest to lowest
SELECT e.emp_id,
       e.F_Name || ' ' || e.L_Name AS full_name,
       FN_NET_SALARY(e.emp_id)     AS net_salary
FROM EMPLOYEE e
ORDER BY net_salary DESC NULLS LAST;

-- Create SP_PROCESS_PAYROLL
CREATE OR REPLACE PROCEDURE SP_PROCESS_PAYROLL (
    p_job_id   IN NUMBER,
    p_pay_date IN DATE
) AS
    CURSOR c_employees IS
        SELECT e.emp_id, e.salary_id, sb.amount,
               NVL(sb.bons, 0)  AS bonus,
               sb.amount / 30   AS daily_rate
        FROM EMPLOYEE e
        JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
        WHERE e.job_id = p_job_id;

    v_emp      c_employees%ROWTYPE;
    v_leave    NUMBER;
    v_total    NUMBER;
    v_inserted NUMBER := 0;
BEGIN
    OPEN c_employees;
    LOOP
        FETCH c_employees INTO v_emp;
        EXIT WHEN c_employees%NOTFOUND;

        -- Count leave records for this employee
        SELECT COUNT(*) INTO v_leave
        FROM LEAVEE WHERE emp_id = v_emp.emp_id;

        -- total_amount = salary + bonus - (leave_count * daily_rate)
        v_total := v_emp.amount + v_emp.bonus
                   - (v_leave * v_emp.daily_rate);

        -- Insert one payroll row
        INSERT INTO PAYROLL
            (pay_id, total_am, report, pay_date, emp_id, salary_id, leave_id)
        VALUES
            (seq_payroll.NEXTVAL, v_total, 'Auto payroll run',
             p_pay_date, v_emp.emp_id, v_emp.salary_id, NULL);

        v_inserted := v_inserted + 1;
    END LOOP;
    CLOSE c_employees;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Payroll processed: ' || v_inserted
        || ' records inserted for job_id ' || p_job_id || '.');

EXCEPTION
    WHEN OTHERS THEN
        IF c_employees%ISOPEN THEN CLOSE c_employees; END IF;
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Payroll error: ' || SQLERRM);
END SP_PROCESS_PAYROLL;

  TASK 4 │ Procedure — Raise Salary with Audit
────────────────────────────────────────────────────────────

-- Create SALARY_AUDIT table
CREATE TABLE SALARY_AUDIT (
    audit_id   NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    emp_id     NUMBER        NOT NULL,
    old_amount NUMBER(10,2)  NOT NULL,
    new_amount NUMBER(10,2)  NOT NULL,
    changed_by VARCHAR2(50)  DEFAULT USER,
    changed_at TIMESTAMP     DEFAULT SYSTIMESTAMP
);


-- Create SP_RAISE_SALARY
CREATE OR REPLACE PROCEDURE SP_RAISE_SALARY (
    p_emp_id     IN NUMBER,
    p_percentage IN NUMBER
) AS
    v_salary_id NUMBER;
    v_old_amt   NUMBER;
    v_new_amt   NUMBER;
BEGIN
    -- Retrieve current salary
    SELECT e.salary_id, sb.amount
    INTO v_salary_id, v_old_amt
    FROM EMPLOYEE e
    JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
    WHERE e.emp_id = p_emp_id;

    -- Calculate and apply raise
    v_new_amt := ROUND(v_old_amt * (1 + p_percentage / 100), 2);

    UPDATE SALARY_BONUS
    SET amount = v_new_amt
    WHERE salary_id = v_salary_id;

    -- Write audit record
    INSERT INTO SALARY_AUDIT (emp_id, old_amount, new_amount)
    VALUES (p_emp_id, v_old_amt, v_new_amt);

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Salary raised for emp_id ' || p_emp_id
        || ': ' || v_old_amt || ' -> ' || v_new_amt);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Error: emp_id ' || p_emp_id
            || ' not found. No update performed.');
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END SP_RAISE_SALARY;
/

-- Package Body (private helper + all four subprograms)
CREATE OR REPLACE PACKAGE BODY PKG_EMPLOYEE_MGMT AS

    -- Private: email-validation helper (not callable externally)
    FUNCTION FN_VALIDATE_EMAIL (p_email IN VARCHAR2) RETURN BOOLEAN AS
        v_cnt NUMBER;
    BEGIN
        SELECT COUNT(*) INTO v_cnt
        FROM EMP_EMAIL WHERE LOWER(emp_email) = LOWER(p_email);
        RETURN v_cnt = 0;
    END FN_VALIDATE_EMAIL;

    -- SP_ADD_EMPLOYEE
    PROCEDURE SP_ADD_EMPLOYEE (
        p_emp_pass  IN VARCHAR2, p_age       IN NUMBER,
        p_gender    IN CHAR,     p_F_Name    IN VARCHAR2,
        p_L_Name    IN VARCHAR2, p_street    IN VARCHAR2,
        p_zip_code  IN VARCHAR2, p_job_id    IN NUMBER,
        p_salary_id IN NUMBER,   p_email     IN VARCHAR2
    ) AS
        v_emp_id           NUMBER;
        ex_duplicate_email EXCEPTION;
    BEGIN
        IF NOT FN_VALIDATE_EMAIL(p_email) THEN
            RAISE ex_duplicate_email;
        END IF;
        v_emp_id := seq_employee.NEXTVAL;
        INSERT INTO EMPLOYEE (emp_id, emp_pass, age, gender, F_Name, L_Name,
                              street, zip_code, job_id, salary_id)
        VALUES (v_emp_id, p_emp_pass, p_age, p_gender, p_F_Name, p_L_Name,
                p_street, p_zip_code, p_job_id, p_salary_id);
        INSERT INTO EMP_EMAIL (emp_email, emp_id)
        VALUES (LOWER(p_email), v_emp_id);
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Employee ' || p_F_Name || ' ' || p_L_Name
            || ' added with ID ' || v_emp_id || '.');
    EXCEPTION
        WHEN ex_duplicate_email THEN
            DBMS_OUTPUT.PUT_LINE('Error: email ' || p_email || ' already in use.');
        WHEN OTHERS THEN
            ROLLBACK;
            DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
    END SP_ADD_EMPLOYEE;

    -- FN_NET_SALARY
    FUNCTION FN_NET_SALARY (p_emp_id IN NUMBER) RETURN NUMBER AS
        v_net NUMBER;
    BEGIN
        SELECT sb.amount + NVL(sb.bons, 0)
        INTO v_net
        FROM EMPLOYEE e
        JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
        WHERE e.emp_id = p_emp_id;
        RETURN v_net;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN RETURN NULL;
    END FN_NET_SALARY;

    -- SP_PROCESS_PAYROLL
    PROCEDURE SP_PROCESS_PAYROLL (
        p_job_id IN NUMBER, p_pay_date IN DATE
    ) AS
        CURSOR c_emp IS
            SELECT e.emp_id, e.salary_id, sb.amount,
                   NVL(sb.bons, 0) AS bonus,
                   sb.amount / 30  AS daily_rate
            FROM EMPLOYEE e
            JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
            WHERE e.job_id = p_job_id;
        v_row      c_emp%ROWTYPE;
        v_leave    NUMBER;
        v_total    NUMBER;
        v_inserted NUMBER := 0;
    BEGIN
        OPEN c_emp;
        LOOP
            FETCH c_emp INTO v_row;
            EXIT WHEN c_emp%NOTFOUND;
            SELECT COUNT(*) INTO v_leave FROM LEAVEE WHERE emp_id = v_row.emp_id;
            v_total := v_row.amount + v_row.bonus - (v_leave * v_row.daily_rate);
            INSERT INTO PAYROLL
                (pay_id, total_am, report, pay_date, emp_id, salary_id, leave_id)
            VALUES (seq_payroll.NEXTVAL, v_total, 'Package payroll run',
                    p_pay_date, v_row.emp_id, v_row.salary_id, NULL);
            v_inserted := v_inserted + 1;
        END LOOP;
        CLOSE c_emp;
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Payroll done: '||v_inserted||' records inserted.');
    EXCEPTION
        WHEN OTHERS THEN
            IF c_emp%ISOPEN THEN CLOSE c_emp; END IF;
            ROLLBACK;
            DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
    END SP_PROCESS_PAYROLL;

    -- SP_RAISE_SALARY
    PROCEDURE SP_RAISE_SALARY (
        p_emp_id IN NUMBER, p_percentage IN NUMBER
    ) AS
        v_salary_id NUMBER;
        v_old       NUMBER;
        v_new       NUMBER;
    BEGIN
        SELECT e.salary_id, sb.amount
        INTO v_salary_id, v_old
        FROM EMPLOYEE e
        JOIN SALARY_BONUS sb ON e.salary_id = sb.salary_id
        WHERE e.emp_id = p_emp_id;
        v_new := ROUND(v_old * (1 + p_percentage / 100), 2);
        UPDATE SALARY_BONUS SET amount = v_new WHERE salary_id = v_salary_id;
        INSERT INTO SALARY_AUDIT (emp_id, old_amount, new_amount)
        VALUES (p_emp_id, v_old, v_new);
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Salary updated: ' || v_old || ' -> ' || v_new);
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Error: emp_id '||p_emp_id||' not found.');
        WHEN OTHERS THEN
            ROLLBACK;
            DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
    END SP_RAISE_SALARY;

END PKG_EMPLOYEE_MGMT;
